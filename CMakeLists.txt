cmake_minimum_required(VERSION 3.20)
project(Luxware VERSION 1.1.2 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(Luxware WIN32 src/main.cpp app.rc)
target_include_directories(Luxware PRIVATE src vendor "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(Luxware PRIVATE UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(Luxware PRIVATE user32 gdi32 comctl32 comdlg32 shell32 ole32 uuid dwmapi d3d11 dxgi winmm)
if(MSVC)
  target_compile_options(Luxware PRIVATE /EHsc /utf-8 /W3)
  target_link_options(Luxware PRIVATE /MANIFEST:NO)
else()
  target_compile_options(Luxware PRIVATE -O2)
  target_link_options(Luxware PRIVATE -municode -static -s)
endif()
foreach(file release.json onnxruntime.dll onnxruntime_providers_shared.dll DirectML.dll MODEL-LICENSE.txt ONNXRuntime-LICENSE.txt ONNXRuntime-ThirdPartyNotices.txt DirectML-LICENSE.txt DirectML-LICENSE-CODE.txt)
  add_custom_command(TARGET Luxware POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/${file}" "$<TARGET_FILE_DIR:Luxware>/${file}")
endforeach()
foreach(folder Models Profiles Tests Prerequisites Updates)
  add_custom_command(TARGET Luxware POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/${folder}" "$<TARGET_FILE_DIR:Luxware>/${folder}")
endforeach()
